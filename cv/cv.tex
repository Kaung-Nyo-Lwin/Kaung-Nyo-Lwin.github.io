\begin{cvcard}
  {\cventrytitlefont L2BC: Reinforcement Fine-Tuning for Text-to-Visualization\par}
  {\cvorgfont Master's thesis \textperiodcentered{} Asian Institute of Technology}\hfill{\cvdatefont 2026}\par
  {\cvcontextfont L2BC (``Learn To Be Cool''): small code LLMs learn to chart data from rewards that are verified, not judged}\par
  \begin{itemize}
    \item Proposed, to our knowledge, the first reinforcement fine-tuning framework for text-to-visualization with a purely
      rule-based reward: programs are executed, and their rendered charts parsed and verified against ground truth.
    \item Turned the verified reward's expectation (``comfortability'') into a training-state signal that sets GRPO's
      temperature, learning rate and reward shape each epoch, with no LLM or VLM judge in the loop.
    \item Fine-tuned Qwen2.5-Coder 1.5B and 3B with LoRA and TRL on 192 VisEval queries; the 3B model reaches 67.97\%
      single-table and 59.26\% multi-table pass rates in one pass, beating 7B agentic baselines (59.87\%, 43.32\%).
    \item Reward decomposition and ablations (workflow structure, scale, chain-of-thought) show what verification teaches
      and where it stops: unrewarded sort-order errors grow on unseen tasks.
  \end{itemize}
  \vspace{2.5pt}
  \cvpub{K.\,N.~Lwin, T.~Tangtortan, C.~Silpasuwanchai. \textit{L2BC: Reinforcement Fine-Tuning for Text-to-Visualization
    with Rule-Based Rewards over Parsed Chart Schemas.} Manuscript in preparation, 2026.}
\end{cvcard}
